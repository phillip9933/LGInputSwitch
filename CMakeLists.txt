cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
project(LGInputSwitch VERSION 1.2.0 LANGUAGES CXX)
set(LG_BUILD_REVISION "unavailable")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.git")
    execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                    OUTPUT_VARIABLE LG_GIT_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(LG_GIT_REVISION MATCHES "^[0-9a-f]+$")
        set(LG_BUILD_REVISION "${LG_GIT_REVISION}")
        execute_process(COMMAND git diff --quiet HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                        RESULT_VARIABLE LG_GIT_MODIFIED ERROR_QUIET)
        if(NOT LG_GIT_MODIFIED EQUAL 0)
            string(APPEND LG_BUILD_REVISION "-modified")
        endif()
    endif()
elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/SOURCE_REVISION")
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/SOURCE_REVISION" LG_ARCHIVE_REVISION LIMIT_COUNT 1)
    if(LG_ARCHIVE_REVISION MATCHES "^[0-9a-f]+$")
        set(LG_BUILD_REVISION "${LG_ARCHIVE_REVISION}")
    endif()
endif()
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(LG_BUILD_DESKTOP "Build the modern Qt desktop frontend" OFF)
option(LG_REQUIRE_KDE_SHORTCUTS "Fail if native KDE shortcuts cannot be built" OFF)

if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(lginput_core core/input_switcher.cpp)
target_include_directories(lginput_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/core)

if(WIN32)
    enable_language(RC)
endif()

if(LG_BUILD_DESKTOP)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent)
    if(NOT WIN32)
        find_package(KF6GlobalAccel QUIET)
        if(LG_REQUIRE_KDE_SHORTCUTS AND NOT KF6GlobalAccel_FOUND)
            message(FATAL_ERROR "KDE shortcuts require KF6GlobalAccel development files")
        endif()
        if(KF6GlobalAccel_FOUND)
            find_package(Qt6 REQUIRED COMPONENTS DBus)
        endif()
    endif()
    # Qt's Windows entrypoint/runtime uses the dynamic MSVC CRT. Compile the
    # same portable core with that runtime without changing legacy /MT builds.
    add_library(lginput_desktop_core core/input_switcher.cpp)
    if(MSVC)
        set_property(TARGET lginput_desktop_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    add_executable(LGInputSwitchDesktop WIN32 desktop/main.cpp desktop/platform.cpp)
    function(lg_configure_desktop target)
    target_sources(${target} PRIVATE desktop/diagnostics.cpp desktop/capabilities.cpp)
    target_compile_definitions(${target} PRIVATE LG_BUILD_REVISION="${LG_BUILD_REVISION}")
    target_link_libraries(${target} PRIVATE lginput_desktop_core Qt6::Widgets Qt6::Concurrent)
    if(TARGET KF6::GlobalAccel)
        target_compile_definitions(${target} PRIVATE LG_HAVE_KGLOBALACCEL)
        target_link_libraries(${target} PRIVATE KF6::GlobalAccel Qt6::DBus)
    endif()
    if(MSVC)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
    if(WIN32)
        target_sources(${target} PRIVATE app/app_config.cpp app/hotkeys.cpp
            platform/windows/adl_session.cpp platform/windows/adl_command.cpp platform/windows/adl_transport.cpp resource/app.rc)
        target_include_directories(${target} PRIVATE app adl-sdk/include resource)
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_link_libraries(${target} PRIVATE user32 shell32 advapi32 gdi32 dxva2)
    else()
        target_sources(${target} PRIVATE platform/linux/ddcutil_backend.cpp)

    endif()
    endfunction()
    lg_configure_desktop(LGInputSwitchDesktop)
    if(NOT WIN32)
        install(TARGETS LGInputSwitchDesktop RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
        install(FILES desktop/lginputswitch.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
        install(FILES desktop/lginputswitch.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
    endif()
    if(BUILD_TESTING)
        add_executable(diagnostics_test tests/diagnostics.cpp desktop/diagnostics.cpp desktop/capabilities.cpp)
        target_link_libraries(diagnostics_test PRIVATE Qt6::Core)
        target_compile_definitions(diagnostics_test PRIVATE LG_BUILD_REVISION="${LG_BUILD_REVISION}")
        if(WIN32)
            target_compile_definitions(diagnostics_test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
            target_link_libraries(diagnostics_test PRIVATE user32 advapi32 dxva2)
            if(MSVC)
                set_property(TARGET diagnostics_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
            endif()
        endif()
        add_test(NAME diagnostics COMMAND diagnostics_test)
        add_executable(desktop_smoke tests/desktop_smoke.cpp desktop/platform.cpp)
        lg_configure_desktop(desktop_smoke)
        add_test(NAME desktop-smoke COMMAND desktop_smoke)
        set_tests_properties(desktop-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
        if(TARGET KF6::GlobalAccel)
            # Manual live-session probe: registers only an isolated test
            # component and uses a fake monitor backend. No hardware writes.
            add_executable(kde_window_probe tests/kde_window_probe.cpp desktop/platform.cpp)
            lg_configure_desktop(kde_window_probe)
        endif()
    endif()
endif()


if(WIN32)
    add_executable(LGInputSwitch WIN32
        app/main.cpp app/app_tray.cpp app/app_config.cpp app/app_toggle.cpp
        app/hotkeys.cpp app/settings_ui.cpp app/welcome_ui.cpp
        platform/windows/adl_session.cpp platform/windows/adl_command.cpp platform/windows/adl_transport.cpp
        resource/app.rc)
    target_include_directories(LGInputSwitch PRIVATE app external adl-sdk/include resource)
    target_compile_definitions(LGInputSwitch PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(LGInputSwitch PRIVATE lginput_core user32 shell32 advapi32 gdi32)
endif()

    add_executable(lginputswitch cli/main_linux.cpp platform/linux/ddcutil_backend.cpp)

    install(TARGETS lginputswitch RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME release-packaging COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/release_packaging.py)
    add_executable(regression tests/regression.cpp app/app_toggle.cpp app/hotkeys.cpp
        platform/windows/adl_transport.cpp)
    target_link_libraries(regression PRIVATE lginput_core)
    if(NOT WIN32)
        target_include_directories(regression PRIVATE tests/win32)
    endif()
    add_test(NAME regression COMMAND regression)
    add_executable(core_regression tests/core_regression.cpp)
    target_link_libraries(core_regression PRIVATE lginput_core)
    add_test(NAME core COMMAND core_regression)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME packaging COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/packaging.py)
    add_test(NAME linux-cli COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/linux_cli.py $<TARGET_FILE:lginputswitch>)
    endif()
    if(WIN32)
        add_library(fake_adl SHARED tests/fake_adl.cpp)
        add_library(fake_adl_incomplete SHARED tests/fake_adl.cpp)
        target_compile_definitions(fake_adl_incomplete PRIVATE LG_FAKE_ADL_INCOMPLETE)
        add_executable(adl_backend_test tests/adl_backend_windows.cpp
            platform/windows/adl_session.cpp platform/windows/adl_command.cpp)
        target_compile_definitions(adl_backend_test PRIVATE LG_ADL_TESTING)
        target_link_libraries(adl_backend_test PRIVATE lginput_core)
        add_test(NAME adl-backend COMMAND adl_backend_test
            $<TARGET_FILE:fake_adl> $<TARGET_FILE:fake_adl_incomplete>)
        add_executable(startup_test tests/startup_windows.cpp)
        target_link_libraries(startup_test PRIVATE advapi32)
        set_target_properties(startup_test PROPERTIES OUTPUT_NAME "Startup Test")
        add_test(NAME startup COMMAND startup_test)
    endif()
endif()
